# --- PDF viewing ----------------------------------------------------------
# zathura is vimtex's first-class viewer: auto-opens on compile, and both
# forward and inverse SyncTeX search work with no configuration. Wanted on
# every machine that edits LaTeX, which is all of them.
#
# Native rather than Flatpak — a deliberate Tier 2 exception, because inverse
# search has to launch nvim on the host and a sandbox makes that painful for
# no benefit. okular stays installed as part of the KDE spin; it just stops
# being what opens PDFs.
zathura
zathura-pdf-mupdf         # mupdf backend — lighter and faster than poppler

# --- shell tooling --------------------------------------------------------
# eza: ls replacement. Aliased in config/fish/config.fish rather than shadowing
# `ls` outright — scripts and muscle memory that expect real ls behaviour keep
# getting real ls. Icons need a Nerd Font; see the note in config.fish.
eza

# btop: process/resource monitor. Not a preference — waybar's cpu and memory
# modules open it on click (config/waybar/config.jsonc), so on the ThinkPad it
# is a hard dependency of the bar. A comment in that file used to claim btop
# "is already installed", which was an assumption, not a check: it was in no
# package list at all. Found 2026-08-24.
btop

# --- what the scripts in bin/ actually need -------------------------------
# Each of these was invoked by a committed script while appearing in no package
# list, which makes a rebuild fail at the point of use rather than at install
# time. Audited 2026-08-24.

# notify-send. bin/backup-local, bin/backup-verify and bin/pkgwatch all use it
# to escalate a failure to the desktop, and all three guard it with
# `command -v` — so without libnotify they degrade to a silent failure, which
# is the exact outcome those scripts exist to prevent.
libnotify

# wl-clipboard (wl-copy/wl-paste). neovim's clipboard integration uses it on
# every Wayland session, and config/river/init pipes cliphist through it. Both
# packages/bundles/river.txt and a comment in config/fish assumed it was
# "already installed" — true on a KDE spin, but an assumption rather than a
# declaration, and a rebuild does not honour assumptions.
wl-clipboard
